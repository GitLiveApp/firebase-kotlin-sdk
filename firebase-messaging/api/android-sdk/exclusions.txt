# Firebase Android SDK APIs intentionally not mirrored by firebase-messaging.
# @platform members are declared in common code through error-deprecated stand-ins for the Android/JVM type (android.app.Activity, java.util.concurrent.Executor, android.os.Parcel, ...), so common code gets a message naming the type rather than an unresolved reference; they are not counted.
com.google.firebase.messaging.FirebaseMessagingService  // @platform an android.app.Service: received messages are handled by platform code on every platform
com.google.firebase.messaging.FirebaseMessaging#register  // added after firebase-messaging 25.1.2, the version this SDK depends on
com.google.firebase.messaging.FirebaseMessaging#unregister  // added after firebase-messaging 25.1.2, the version this SDK depends on
com.google.firebase.messaging.FirebaseMessaging#deliveryMetricsExportToBigQueryEnabled  // @unavailable Android-only: the Apple and JS SDKs export delivery metrics through their own configuration
com.google.firebase.messaging.FirebaseMessaging#setDeliveryMetricsExportToBigQuery  // @unavailable Android-only, see above
com.google.firebase.messaging.FirebaseMessaging#isNotificationDelegationEnabled  // @unavailable Android-only: proxy notifications are an Android feature
com.google.firebase.messaging.FirebaseMessaging#setNotificationDelegationEnabled  // @unavailable Android-only, see above
com.google.firebase.messaging.RemoteMessage#writeToParcel  // @platform android.os.Parcelable
