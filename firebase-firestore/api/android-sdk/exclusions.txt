# Firebase Android SDK APIs intentionally not mirrored by firebase-firestore.

# Firestore Pipelines have no supported Objective-C API in the iOS SDK: the supported API is Swift, and the Objective-C side is
# the `__`-prefixed FIRPipelineBridge classes, hidden as internal since 13.0.0 (FlutterFire and React Native Firebase build on
# them anyway; firebase/firebase-ios-sdk#17013 asks for them to be formalised). In the pinned JS SDK they are a separate
# preview entry point (firebase/firestore/pipelines).
com.google.firebase.firestore.pipeline.*  // @unavailable Firestore Pipelines have no supported Objective-C API in the iOS SDK (firebase/firebase-ios-sdk#17013) and are a preview entry point in the JS SDK
com.google.firebase.firestore.Pipeline*  // @unavailable Firestore Pipelines have no supported Objective-C API in the iOS SDK (firebase/firebase-ios-sdk#17013) and are a preview entry point in the JS SDK
com.google.firebase.firestore.FirebaseFirestore#pipeline  // @unavailable Firestore Pipelines have no supported Objective-C API in the iOS SDK (firebase/firebase-ios-sdk#17013) and are a preview entry point in the JS SDK
